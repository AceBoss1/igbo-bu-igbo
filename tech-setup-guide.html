Frontend

Next.js 14 with App Router. React components, server-side rendering for SEO, API routes for backend logic. Deployed on Vercel (free tier).

Backend + DB

Firebase handles authentication, Firestore database, and file storage. No separate server needed at launch — Firebase handles everything.

Payments

Paystack processes all Nigerian Naira payments. Free to integrate — only 1.5% + ₦100 per transaction. Supports cards, bank transfer, USSD.

Hosting & CDN

Vercel hosts the Next.js app free (100 GB bandwidth/month). Cloudflare free tier provides CDN and DDoS protection for the domain.

Firestore Security Rules
firestore rulesfirestore.rules
rules_version = '2'; service cloud.firestore { match /databases/{database}/documents { // Helper functions function isAuth() { return request.auth != null; } function isAdmin() { return isAuth() && get(/databases/$(database)/documents/members/$(request.auth.uid)).data.role in ['super_admin','national_exec','state_coord']; } function isOwner(uid) { return request.auth.uid == uid; } // Members — read own, admin reads all match /members/{memberId} { allow read: if isOwner(memberId) || isAdmin(); allow create: if isAuth(); allow update: if isOwner(memberId) || isAdmin(); allow delete: if isAdmin(); } // Public verification — only read specific safe fields match /members/{memberId} { allow get: if resource.data.keys().hasOnly( ['fullName','membershipId','status','chapter','photoURL']); } // Payments — members read own match /payments/{payId} { allow read: if isAuth() && resource.data.userId == request.auth.uid || isAdmin(); allow write: if isAdmin(); // Only server/admin can write payments } // Elections — all authenticated members can read; admin writes match /elections/{elId} { allow read: if isAuth(); allow write: if isAdmin(); } // Votes — write-only for member (no re-reading own vote) match /votes/{voteId} { allow create: if isAuth() && !exists( /databases/$(database)/documents/votes/$(request.auth.uid + '_' + request.resource.data.electionId)); allow read: if isAdmin(); } // Grants, Transfers, Wallet — member reads/writes own match /transfers/{tId} { allow read, create: if isAuth() && request.resource.data.userId == request.auth.uid; allow update: if isAdmin(); } match /wallet/{wId} { allow read: if isAuth() && isOwner(wId); allow write: if isAdmin(); // Only server updates wallet balances } } }

typescriptsrc/lib/auth-helpers.ts
import { signInWithEmailAndPassword, createUserWithEmailAndPassword, sendEmailVerification, sendPasswordResetEmail } from 'firebase/auth'; import { auth, db } from './firebase'; import { doc, getDoc } from 'firebase/firestore'; // LOGIN export async function loginMember(email: string, password: string) { const cred = await signInWithEmailAndPassword(auth, email, password); const member = await getDoc(doc(db, 'members', cred.user.uid)); if (member.data()?.status === 'suspended') throw new Error('Account suspended. Contact admin.'); return { user: cred.user, memberData: member.data() }; } // REGISTER (creates Firebase Auth user + Firestore member doc) export async function registerMember(email: string, password: string, data: MemberData) { const cred = await createUserWithEmailAndPassword(auth, email, password); await sendEmailVerification(cred.user); // Create pending member doc in Firestore await setDoc(doc(db, 'members', cred.user.uid), { ...data, uid: cred.user.uid, status: 'pending', createdAt: serverTimestamp(), role: 'member' }); return cred.user; }

Paystack Webhook (Verify & Update Database)
typescriptsrc/app/api/paystack/webhook/route.ts
import { NextRequest, NextResponse } from 'next/server'; import crypto from 'crypto'; import { adminDb } from '@/lib/firebase-admin'; export async function POST(req: NextRequest) { const secret = process.env.PAYSTACK_SECRET_KEY!; const hash = req.headers.get('x-paystack-signature'); const body = await req.text(); // Verify webhook signature (CRITICAL — prevents fake webhooks) const expected = crypto.createHmac('sha512', secret).update(body).digest('hex'); if (hash !== expected) return NextResponse.json({ error: 'Invalid signature' }, { status: 401 }); const event = JSON.parse(body); if (event.event === 'charge.success') { const { reference, amount, metadata } = event.data; // Save payment to Firestore await adminDb.collection('payments').doc(reference).set({ reference, amount: amount / 100, amountKobo: amount, status: 'success', userId: metadata.userId, memberId: metadata.memberId, type: metadata.paymentType, paidAt: new Date(), channel: event.data.channel }); // If registration fee — activate pending member if (metadata.paymentType === 'registration') { await adminDb.collection('members').doc(metadata.userId).update({ status: 'pending_approval' // Admin still needs to approve }); } } return NextResponse.json({ received: true }); }
Set your webhook URL in Paystack Dashboard → Settings → API Keys & Webhooks → Webhook URL: https://igbobuigbo.org.ng/api/paystack/webhook
